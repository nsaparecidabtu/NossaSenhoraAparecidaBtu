enum StaffRole {
  SUPER_ADMIN     // Dona Maria — acesso total, é quem delega o resto
  MINISTRY_LEADER // acesso restrito ao que foi delegado via `permissions`
}

// Capacidades concedidas individualmente pela Dona Maria — a lista cresce
// conforme surgir necessidade real (evita repetir o erro do enum de
// roles genéricos sem lógica por trás, do outro projeto).
enum StaffPermission {
  VIEW_PRAYER_REQUESTS // ex: líder da Acolhida — vê pedidos de oração, inclusive os sigilosos
  MANAGE_TITHE_RAFFLE  // ex: coordenadora do Dízimo — publica sorteio mensal
  MANAGE_EVENTS
  MANAGE_GALLERY
  MANAGE_MASS_SCHEDULE
  MANAGE_MINISTRIES
  MANAGE_FAQ
}

model User {
  id            String    @id @default(cuid())
  name          String?
  email         String?   @unique
  emailVerified DateTime?
  image         String?

  // null = paroquiano comum (só pode fazer pedidos/solicitações).
  // SUPER_ADMIN = Dona Maria, acesso total, sem precisar checar `permissions`.
  // MINISTRY_LEADER = acesso só ao que estiver em `permissions`.
  staffRole   StaffRole?
  ministryId  String?
  ministry    Ministry?         @relation(fields: [ministryId], references: [id])
  permissions StaffPermission[] @default([])

  accounts Account[]
  sessions Session[]
  contactRequests ContactRequest[]

  createdAt DateTime @default(now())
}

model Account {
  id                String  @id @default(cuid())
  userId            String
  type              String
  provider          String
  providerAccountId String
  refresh_token     String? @db.Text

  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String? @db.Text
  session_state     String?

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerAccountId])
}

// src/actions/massSchedule.ts
'use server'

import { prisma } from '@/lib/prisma'
import { revalidatePath } from 'next/cache'
import { requirePermission } from '@/lib/permissions'

export async function createMassSchedule(_prevState: unknown, formData: FormData) {
  try {
    await requirePermission('MANAGE_MASS_SCHEDULE')

    const label = formData.get('label') as string
    const timesRaw = formData.get('times') as string
    const order = Number(formData.get('order') ?? 0)

    if (!label || !timesRaw) throw new Error('Preencha o rótulo e ao menos um horário.')

    const times = timesRaw
      .split(',')
      .map((t) => t.trim())
      .filter(Boolean)

    if (times.length === 0) throw new Error('Informe ao menos um horário válido.')

    const schedule = await prisma.massSchedule.create({
      data: { label, times, order },
    })

    revalidatePath('/admin/horarios')
    revalidatePath('/')
    return { success: true, data: schedule }
  } catch (error: any) {
    return { success: false, error: error.message || 'Falha ao criar horário.' }
  }
}

export async function deleteMassSchedule(id: string) {
  try {
    await requirePermission('MANAGE_MASS_SCHEDULE')
    await prisma.massSchedule.delete({ where: { id } })
    revalidatePath('/admin/horarios')
    revalidatePath('/')
    return { success: true }
  } catch (error: any) {
    return { success: false, error: error.message || 'Falha ao excluir.' }

// src/lib/liturgical.ts

// Calcula o tempo litúrgico atual a partir da data da Páscoa (algoritmo
// de Gauss). Aproximação cobrindo os grandes tempos (Advento, Natal,
// Quaresma, Tríduo, Páscoa, Tempo Comum) — não cobre toda solenidade ou
// festa de santo específica, o que é aceitável pro indicador simples
// que foi pedido ("hoje é tempo comum/verde").

export type LiturgicalSeason = {
  name: string
  color: 'purple' | 'white' | 'green' | 'red'
  colorHex: string
}

function addDays(date: Date, days: number): Date {
  const d = new Date(date)
  d.setDate(d.getDate() + days)
  return d
}

function isBetween(date: Date, start: Date, end: Date): boolean {
  return date >= start && date <= end
}

function getEasterDate(year: number): Date {
  const a = year % 19
  const b = Math.floor(year / 100)
  const c = year % 100
  const d = Math.floor(b / 4)
  const e = b % 4
  const f = Math.floor((b + 8) / 25)
  const g = Math.floor((b - f + 1) / 3)
  const h = (19 * a + b - d - g + 15) % 30
  const i = Math.floor(c / 4)
  const k = c % 4
  const l = (32 + 2 * e + 2 * i - h - k) % 7
  const m = Math.floor((a + 11 * h + 22 * l) / 451)
  const month = Math.floor((h + l - 7 * m + 114) / 31)
  const day = ((h + l - 7 * m + 114) % 31) + 1
  return new Date(year, month - 1, day)
}

// 1º domingo do Advento = o domingo entre 27/nov e 3/dez (sempre existe exatamente um)
function getFirstSundayOfAdvent(year: number): Date {
  for (let day = 27; day <= 30; day++) {
    const d = new Date(year, 10, day)
    if (d.getDay() === 0) return d
  }
  for (let day = 1; day <= 3; day++) {
    const d = new Date(year, 11, day)
    if (d.getDay() === 0) return d
  }
  throw new Error('Não foi possível calcular o Advento')
}

// Batismo do Senhor ≈ domingo seguinte a 6/jan (aproximação)
function getBaptismOfTheLord(year: number): Date {
  for (let day = 7; day <= 13; day++) {
    const d = new Date(year, 0, day)
    if (d.getDay() === 0) return d
  }
  throw new Error('Não foi possível calcular')
}

const COLORS = {
  purple: '#5b2a86',
  white: '#d4a017',
  green: '#2f6b3a',
  red: '#a11d1d',
} as const

export function getLiturgicalSeason(date: Date = new Date()): LiturgicalSeason {
  const year = date.getFullYear()
  const easter = getEasterDate(year)
  const ashWednesday = addDays(easter, -46)
  const palmSunday = addDays(easter, -7)
  const holyThursday = addDays(easter, -3)
  const pentecost = addDays(easter, 49)
  const adventStart = getFirstSundayOfAdvent(year)
  const christmasStartThisYear = new Date(year, 11, 25)
  const christmasEnd = getBaptismOfTheLord(year)

  if (isBetween(date, palmSunday, addDays(holyThursday, -1))) {
    return { name: 'Semana Santa', color: 'purple', colorHex: COLORS.purple }
  }
  if (isBetween(date, holyThursday, addDays(easter, -1))) {
    return { name: 'Tríduo Pascal', color: 'red', colorHex: COLORS.red }
  }
  if (isBetween(date, ashWednesday, addDays(palmSunday, -1))) {
    return { name: 'Quaresma', color: 'purple', colorHex: COLORS.purple }
  }
  if (isBetween(date, easter, pentecost)) {
    return { name: 'Tempo Pascal', color: 'white', colorHex: COLORS.white }
  }
  if (isBetween(date, christmasStartThisYear, new Date(year, 11, 31))) {
    return { name: 'Natal', color: 'white', colorHex: COLORS.white }
  }
  if (isBetween(date, new Date(year, 0, 1), christmasEnd)) {
    return { name: 'Natal', color: 'white', colorHex: COLORS.white }
  }
  if (isBetween(date, adventStart, new Date(year, 11, 24))) {
    return { name: 'Advento', color: 'purple', colorHex: COLORS.purple }
  }
  return { name: 'Tempo Comum', color: 'green', colorHex: COLORS.green }
}

import { auth } from '@/auth'
import type { StaffPermission } from '@prisma/client'

export async function requirePermission(permission: StaffPermission) {
  const session = await auth()
  if (!session?.user) throw new Error('Não autenticado.')

  if (session.user.staffRole === 'SUPER_ADMIN') return session

  if (
    session.user.staffRole === 'MINISTRY_LEADER' &&
    session.user.permissions.includes(permission)
  ) {
    return session
  }

  throw new Error('Acesso negado.')
}

export async function requireSuperAdmin() {
  const session = await auth()
  if (session?.user?.staffRole !== 'SUPER_ADMIN') throw new Error('Acesso negado.')
  return session
}
